import pandas as pd from pyvis.network import Network import os # 1. 读取数据 file_path = "/Users/yalujiyuxiaotiqin/Desktop/39-Satellite Art.xlsx" df = pd.read_excel(file_path) # 2. 清洗国家 def clean_country(v): if pd.isna(v): return [] s = str(v).strip() if s == "" or s.lower() in ["na", "nan"]: return [] parts = [p.strip() for p in s.replace(",", "/").split("/") if p.strip()] out = [] for p in parts: up = p.upper() if up in ["USA", "UK", "UAE", "EU", "ESA"]: out.append(up) else: out.append(p.title()) return out df["Country_List"] = df["Country"].apply(clean_country) # 3. 清洗轨道 / 空间位置 def clean_orbit(v): if pd.isna(v): return None s = str(v).strip() if s == "" or s.lower() in ["na", "nan"]: return None low = s.lower() if "iss" in low: return "ISS" if "moon" in low: return "Moon" if "deep" in low: return "Deep Space" if "geo" in low: return "GEO" if "meo" in low: return "MEO" if "leo" in low: return "LEO" return s.title() df["Orbit"] = df["Spatial_Location"].apply(clean_orbit) # 4. 初始化动态网络 net = Network( height="800px", width="100%", bgcolor="#ffffff", font_color="#000000", directed=True ) net.set_options(""" var options = { "nodes": { "shape": "box", "margin": 8, "borderWidth": 1, "shadow": { "enabled": true, "color": "rgba(0,0,0,0.25)", "size": 8, "x": 2, "y": 2 }, "font": { "face": "Helvetica", "size": 14 } }, "edges": { "color": {"color": "#cccccc"}, "smooth": { "enabled": true, "type": "continuous" }, "arrows": { "to": {"enabled": true, "scaleFactor": 0.5} } }, "interaction": { "dragNodes": true, "dragView": true, "zoomView": true, "hover": true, "multiselect": true, "tooltipDelay": 120 }, "physics": { "enabled": true, "stabilization": false, "barnesHut": { "gravitationalConstant": -2500, "centralGravity": 0.1, "springLength": 160, "springConstant": 0.02, "avoidOverlap": 0.3 }, "timestep": 0.5, "minVelocity": 0.1 } } """) artists_added = set() countries_added = set() orbits_added = set() # 5. 添加节点和边：Artist -> Country -> Orbit for _, row in df.iterrows(): creator_raw = row["Creator"] if pd.isna(creator_raw): continue artist = str(creator_raw).strip() if artist == "" or artist.lower() == "nan": continue countries = row["Country_List"] orbit = row["Orbit"] if artist not in artists_added: net.add_node( artist, label=artist, group="Artist", color="#f5f5f5", title=f"Artist: {artist}" ) artists_added.add(artist) for c in countries: if c not in countries_added: net.add_node( c, label=c, group="Country", color="#dce9ff", title=f"Country: {c}" ) countries_added.add(c) net.add_edge(artist, c) if orbit: if orbit not in orbits_added: net.add_node( orbit, label=orbit, group="Orbit", color="#ffe6c7", title=f"Orbit/Location: {orbit}" ) orbits_added.add(orbit) net.add_edge(c, orbit) # 6. 导出 HTML 并插入“下载图片”按钮（从 canvas 导出 PNG） output_path = os.path.expanduser( "~/Desktop/satellite_art_semantic_ACO_dynamic.html" ) net.write_html(output_path) with open(output_path, "r", encoding="utf-8") as f: html = f.read() download_js = """ """ html = html.replace("", download_js) with open(output_path, "w", encoding="utf-8") as f: f.write(html) print("已生成带下载按钮的动态语义图：", output_path)
