// 保存图像为 PNG 文件 document.getElementById("saveBtn").onclick = function () { var canvas = document.getElementsByTagName("canvas")[0]; if (!canvas) { alert("未找到图像区域，请稍后再试。"); return; } var link = document.createElement("a"); link.href = canvas.toDataURL("image/png"); link.download = "country_orbit_artist_network.png"; link.click(); };
